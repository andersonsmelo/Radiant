# injecoes_a4.py (rodado em radiant-app/)
import subprocess, pathlib
f = pathlib.Path('src/features/review/hooks/useReview.ts')
INJ = [
  ('avança em todo item, sem esperar o último da lição', 'agrega os itens da lição',
   "        if (queue.slice(index + 1).some(item => item.lessonId === lessonId)) return;\n", ""),
  ('não recomeça o agregado ao recarregar a fila', 'recarregar a fila recomeça',
   "            legacyTally.current.clear();\n", ""),
  ('conta a resposta antes de ela ser gravada', 'não foi gravada',
   "            let duplicate = false;\n",
   "            let duplicate = false;\n            const early = legacyTally.current.get(currentItem.lessonId) ?? { total: 0, correct: 0 };\n"
   "            early.total += 1; legacyTally.current.set(currentItem.lessonId, early);\n"),
]
out = []
for nome, teste, a, b in INJ:
    orig = f.read_text(encoding='utf8')
    n = orig.count(a); assert n == 1, (nome, n)
    f.write_text(orig.replace(a, b, 1), encoding='utf8')
    try:
        r = subprocess.run(['npx', 'jest', '--runInBand', 'src/features/review/hooks/useReview.test.ts', '-t', teste], capture_output=True, text=True)
    finally:
        f.write_text(orig, encoding='utf8')
    txt = r.stdout + r.stderr
    linhas = [l.rstrip() for l in txt.splitlines() if ('✕' in l or '✓' in l or l.startswith('Tests:'))]
    out.append(f"## {nome} (âncora: {n} ocorrência; teste: \"{teste}\"; saída {r.returncode})\n" + "\n".join(linhas[:8]))
print("\n\n".join(out))
