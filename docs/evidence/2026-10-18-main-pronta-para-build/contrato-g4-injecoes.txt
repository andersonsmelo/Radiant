# injecoes_g4.py
import subprocess, pathlib, sys
fase = sys.argv[1]
f = pathlib.Path('.maestro/g4-native-evidence.yaml')
alvo = ("- scrollUntilVisible:\n    element: Começar revisão\n    direction: DOWN\n"
        "    centerElement: false\n    visibilityPercentage: 80\n")
INJ = [
  ('sem régua', alvo, alvo.replace("    visibilityPercentage: 80\n", "")),
  ('rolagem aninhada sob guarda', alvo,
   "- runFlow:\n    when:\n      notVisible: Começar revisão\n    commands:\n"
   "      - scrollUntilVisible:\n          element: Começar revisão\n          direction: DOWN\n"
   "          centerElement: false\n          visibilityPercentage: 80\n"),
]
out = [f"# fase: {fase}"]
for nome, a, b in INJ:
    orig = f.read_text(encoding='utf8')
    n = orig.count(a); assert n == 1, (nome, n)
    f.write_text(orig.replace(a, b, 1), encoding='utf8')
    try:
        r = subprocess.run(['npm', 'run', '-s', 'test:maestro-contract'], capture_output=True, text=True)
    finally:
        f.write_text(orig, encoding='utf8')
    linhas = [l.strip() for l in r.stdout.splitlines()
              if l.startswith('not ok') or l.startswith('# pass') or l.startswith('# fail')
              or ('g4-native-evidence.yaml' in l and 'scrollUntilVisible' in l)]
    out.append(f"## {nome} (âncora: {n} ocorrência, saída {r.returncode})\n" + "\n".join(linhas[:8]))
print("\n\n".join(out))
